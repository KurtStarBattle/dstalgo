namespace FLORESWinFormExercise
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        // odd or even
        private void button1_Click(object sender, EventArgs e)
        {
            int number1 = Convert.ToInt32(textBox1.Text);

            if (number1 % 2 == 0)
            {
                lblResult1.Text = "Result: Even";
            }
            else
            {
                lblResult1.Text = "Result: Odd";
            }
        }

        // positive or negative
        private void button2_Click(object sender, EventArgs e)
        {
            int number2 = Convert.ToInt32(textBox2.Text);

            if (number2 >= 0)
            {
                lblResult2.Text = "Result: Positive";
            }
            else
            {
                lblResult2.Text = "Result: Negative";
            }
        }

    }
}
