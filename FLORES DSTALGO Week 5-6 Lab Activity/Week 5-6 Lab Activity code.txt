        // O(log n)
        private void button4_Click(object sender, EventArgs e) // binary search
        {
            int[] numbers = { 10, 20, 30, 40, 50 };
            int i;
            if (!int.TryParse(textBox4.Text, out i))
            {
                button4result.Text = "! Invalid Input !";
                return;
            }

            int left = 0, right = numbers.Length - 1, count = 0;
            while (left <= right)
            {
                int mid = (left + right) / 2;
                count++;
                if (numbers[mid] == i)
                {
                    button4result.Text = "Result 1: " + i + " found in " + count + " steps";
                    return;
                }
                if (numbers[mid] < i)
                    left = mid + 1;
                else
                    right = mid - 1;
            }
            button4result.Text = "Result 1: " + i + " not found after " + count + " steps";
        }

        private int Fibonacci(int n, ref int count)
        {
            count++;
            if (n <= 1)
                return n;

            return Fibonacci(n - 1, ref count) + Fibonacci(n - 2, ref count);
        }

        private int Knapsack(int capacity, int n)
        {
            int[] weights = { 2, 3, 4, 5 };
            int[] values = { 3, 4, 5, 6 };

            if (n == 4 || capacity == 0)
                return 0;

            if (weights[n] > capacity)
                return Knapsack(capacity, n + 1);

            return Math.Max(
                Knapsack(capacity, n + 1),
                values[n] + Knapsack(capacity - weights[n], n + 1)
            );
        }
